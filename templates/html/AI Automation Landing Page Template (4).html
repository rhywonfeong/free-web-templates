agent-deploy.ts 
import { Agent } from '@aia/sdk';

const workflow = new Agent({
  name: 'Data_Pipeline_Alpha',
  model: 'nexus-v2-turbo',
  autonomous: true,
});

// Initialize connection
await workflow.connect(process.env.DB_URI);

// Define trigger event
workflow.on('record.created', async (data) => {
  const insight = await workflow.analyze(data);
  return workflow.dispatch(insight);
});

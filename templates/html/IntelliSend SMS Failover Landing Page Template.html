import { IntelliSend } from '@intellisend/node';

const client = new IntelliSend(process.env.API_KEY);

// Intelligence layer handles provider selection
const response = await client.messages.create({
  to: '+15550123456',
  body: 'Your verification code is: 4829',
  strategy: 'lowest_latency', // Optimized routing
  fallback: ['twilio', 'aws_sns', 'messagebird']
});

console.log(response.provider); // "aws_sns"

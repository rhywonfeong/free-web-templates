import { Workflow, Agent } from '@nexus/sdk';

 

const agent = new Agent({

model: 'gpt-4-turbo',

tools: [Search, Calculator]

});

 

// Deploy workflow

await Workflow.create({

trigger: 'webhook',

steps: [agent.run()]

});

Smart Contract Oracles

Trigger on-chain governance based on verified off-chain signatures.

Registry.sol CLUSTER POLICY 
contract ArtifactRegistry {
  mapping(bytes32 => address) public owners;

  function register(bytes32 hash) external {
    require(verifyDID(msg.sender));
    owners[hash] = msg.sender;
    emit ArtifactRegistered(hash, msg.sender);
  }
}

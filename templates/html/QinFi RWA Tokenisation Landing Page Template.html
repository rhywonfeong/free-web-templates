mintAsset.ts 
import { QinFiSDK, AssetType } from '@qinfi/sdk';

// Initialize SDK
const qinfi = new QinFiSDK({
  apiKey: process.env.QINFI_KEY,
  network: 'ethereum'
});

// Tokenize Asset
async function createTokenizedAsset() {
  const asset = await qinfi.assets.create({
    type: AssetType.REAL_ESTATE,
    valuation: 5000000, // $5M USD
    symbol: 'Q-NY55',
    compliance: { kycRequired: true }
  });

  return asset.contractAddress;
}

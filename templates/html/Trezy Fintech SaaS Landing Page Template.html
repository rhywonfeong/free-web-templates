const trezy = require('trezy-node')('sk_test_...');

// Create an instant cross-border transfer
const transfer = await trezy.transfers.create({
  amount: 500000, // 5,000.00
  currency: 'EUR',
  destination: 'acct_1N...',
  source_currency: 'USD',
  metadata: {
    invoice_id: 'INV-4829'
  }
});

console.log(transfer.status); // 'processing'

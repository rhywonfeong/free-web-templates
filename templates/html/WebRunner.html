from browser_use_sdk import BrowserUse

# Initialize the client
client = BrowserUse(api_key="bu_sk_...")

# Define the task

    task="Search for top 10 Hacker News posts",
    llm="browser-use-llm",
    stealth=True
)

# Get structured results
result = task.complete()

import { Parser } from '@parser/sdk';

// Initialize client
const client = new Parser(process.env.API_KEY);

async function scrapePricing() {
  const results = await client.extract({
    urls: ['https://competitor.com/pricing'],
    prompt: 'Extract all plan names and monthly costs'
  });

  console.log(results.data);
}

scrapePricing();

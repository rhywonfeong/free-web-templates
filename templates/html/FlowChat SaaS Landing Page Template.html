webhook-handler.ts 
import { SuperClient } from '@superchat/sdk';

// Initialize client
const client = new SuperClient({
  apiKey: process.env.FLOW_API_KEY
});

// Listen for critical alerts
client.webhooks.on('alert.critical', async (event) => {
  const { service, error, timestamp } = event.data;
  
  await client.channels.send('#ops-emergency', {
    text: `🚨 Critical: ${service} is down`,
    blocks: [
      { type: 'section', content: error.message },
      { type: 'action', label: 'Rollback', id: 'rollback_btn' }
    ]
  });
});

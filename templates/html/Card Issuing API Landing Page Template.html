# Issue a new virtual card
response = client.cards.create(
  type="virtual",
  currency="KZT",
  limit=500000,
  controls={
    "mcc_whitelist": ["software", "advertising"],
    "single_use": False
  }
)

print(f"Card Issued: {response.id}")
print(f"Status: {response.status}")

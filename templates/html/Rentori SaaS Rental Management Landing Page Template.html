Jednoduché API

Rychlý start s tokenem a webhooky pro synchronizaci událostí.

# Autorizace
curl -X POST https://api.rentori.example/v1/auth \
  -H "Content-Type: application/json" \
  -d '{ "apiKey": "YOUR_API_KEY" }'

# Získání položek
curl https://api.rentori.example/v1/items \
  -H "Authorization: Bearer <TOKEN>"
// Rezervace přes REST
fetch('https://api.rentori.example/v1/reservations', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'Authorization': 'Bearer <TOKEN>'
  },
  body: JSON.stringify({
    itemId: 'sku_123',
    from: '2025-11-01',
    to: '2025-11-05',
    customer: { name: 'Jan Novak', email: 'jan@example.com' }
  })
}).then(r => r.json()).then(console.log)

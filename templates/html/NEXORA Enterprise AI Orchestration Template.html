Deterministic Fallbacks

import nexora
from nexora.agents import LLMAgent
from nexora.tools import SQLQuery, SendEmail

# Initialize enterprise client with strict governance
client = nexora.Client(api_key="nx_...", env="production")

# Define an agent with specific boundaries
analyst_agent = LLMAgent(
    name="Financial_Analyst",
    model="gpt-4-turbo",
    tools=[SQLQuery(db="finance_dw")],
    max_retries=3,
    timeout_ms=5000
)

# Create workflow graph
workflow = client.create_workflow(name="Quarterly_Report")
workflow.add_node(analyst_agent)
workflow.deploy()
                                    
import { NexoraClient, LLMAgent } from '@nexora/sdk';
import { SQLQueryTool } from '@nexora/tools/sql';

// Initialize enterprise client
const client = new NexoraClient({ apiKey: process.env.NEXORA_KEY });

const analystAgent = new LLMAgent({
  name: 'Financial_Analyst',
  model: 'gpt-4-turbo',
  tools: [new SQLQueryTool({ db: 'finance_dw' })],
  config: { maxRetries: 3, timeoutMs: 5000 }
});

const workflow = await client.createWorkflow('Quarterly_Report');
workflow.addNode(analystAgent);
await workflow.deploy();
                                    
package main

import (
    "github.com/nexora-io/sdk-go/nexora"
    "github.com/nexora-io/sdk-go/tools"
)

func main() {
    client := nexora.NewClient(nexora.Options{APIKey: "nx_..."})

    analystAgent := nexora.NewLLMAgent(nexora.AgentConfig{
        Name:  "Financial_Analyst",
        Model: "gpt-4-turbo",
        Tools: []nexora.Tool{tools.NewSQLQuery("finance_dw")},
        Retry: 3,
    })

    wf := client.CreateWorkflow("Quarterly_Report")
    wf.AddNode(analystAgent)
    wf.Deploy()
}

"Top 5 users by revenue active last week"
SELECT u.id, u.email, SUM(o.total) as revenue
FROM public.users u
JOIN public.orders o ON u.id = o.user_id
WHERE u.status = 'active'
AND o.created_at > NOW() - INTERVAL '7 days'
GROUP BY u.id
ORDER BY revenue DESC
LIMIT 5;

import { Empatech } from '@empatech/node';

// Initialize the client
const client = new Empatech(process.env.API_KEY);

export async function checkAccess(req, res) {
  const { userId, feature } = req.body;

  // Check entitlement in real-time
  const access = await client.entitlements.can({
    userId,
    feature
  });

  if (!access.granted) {
    return res.status(403).json({ error: 'Upgrade required' });
  }

  next();
}

/**
 * Daily Sync Workflow
 * Orchestrates CRM data to analytics warehouse
 */
import { Workflow } from '@orchestrator/core';
import { crm, warehouse } from './integrations';

export default new Workflow('daily-sync', {
  trigger: 'cron(0 0 * * *)',
  handler: async (ctx) => {
    const users = await crm.getUsers({ 
      updatedAfter: ctx.lastRun 
    });

    // Transform data seamlessly
    const formatted = users.map(u => ({
      id: u.id,
      lifetimeValue: u.ltv,
    }));

    await warehouse.batchInsert('users', formatted);
    
    return { syncedCount: users.length };
  }
});

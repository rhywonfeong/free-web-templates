policy.ts
import { Policy, Rules } from '@neonshield/sdk' ;
const prodProtection = new Policy({
name: 'production-lockdown' ,
target: 'aws:us-east-1' ,
rules: [
Rules. NoPublicBuckets (),
Rules. EnforceMFA ({ strict: true }),
Rules. LimitOpenPorts ([ 80 , 443 ])
],
onViolation: 'block-and-alert'
});
// Deploy policy to active shield
await prodProtection. deploy ();

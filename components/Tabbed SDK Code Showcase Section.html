Infrastructure as Code

Define your event mesh declaratively. Commit your routing rules to Git, test locally, and deploy globally with a single command.

import { EventClient } from '@truenorth/client';

const client = new EventClient({
apiKey: process.env.TRUENORTH_API_KEY,
environment: 'production'
});

// Publish an event to the global mesh
await client.events.publish({
topic: 'user.created',
payload: {
userId: 'usr_9x8f2',
email: 'alex@example.com',
plan: 'enterprise'
},
options: {
idempotencyKey: 'req_9012mf'
}
});
import (
"github.com/truenorth/client-go"
"os"
)

client := truenorth.NewClient(
truenorth.WithAPIKey(os.Getenv("TRUENORTH_API_KEY")),
truenorth.WithEnvironment("production"),
)

// Publish an event to the global mesh
err := client.Events.Publish(&truenorth.Event{
Topic: "user.created",
Payload: map[string]interface{}{
"userId": "usr_9x8f2",
"email": "alex@example.com",
"plan": "enterprise",
},
Options: truenorth.PublishOptions{
IdempotencyKey: "req_9012mf",
},
})
